(function(){
  var state = { headcount'', sectors[], reqs[], notes'' };
  var currentStep = 1;
  var totalSteps = 4;

  function setStep(n) {
     Update step panels
    document.querySelectorAll('.intake-step').forEach(function(el){ el.classList.remove('active'); });
    var panel = document.getElementById('step-'+n);
    if (panel) panel.classList.add('active');

     Update progress indicators
    for (var i=1; i=totalSteps; i++) {
      var ps = document.getElementById('ps-'+i);
      if (!ps) continue;
      ps.classList.remove('active','done');
      if (i  n) ps.classList.add('done');
      else if (i === n) ps.classList.add('active');
    }
    currentStep = n;
    window.scrollTo({ top document.querySelector('.intake-progress').offsetTop - 80, behavior'smooth' });
  }

   Step 1 → 2
  document.getElementById('next-1').addEventListener('click', function(){
    var selected = document.querySelector('input[name=headcount]checked');
    document.getElementById('err-1').classList.remove('visible');
    if (!selected) { document.getElementById('err-1').classList.add('visible'); return; }
    state.headcount = selected.value;
    setStep(2);
  });

   Step 2 → 3
  document.getElementById('back-2').addEventListener('click', function(){ setStep(1); });
  document.getElementById('next-2').addEventListener('click', function(){
    var checked = document.querySelectorAll('input[name=sector]checked');
    document.getElementById('err-2').classList.remove('visible');
    if (checked.length === 0) { document.getElementById('err-2').classList.add('visible'); return; }
    state.sectors = Array.from(checked).map(function(c){ return c.value; });
    setStep(3);
  });

   Step 3 → 4
  document.getElementById('back-3').addEventListener('click', function(){ setStep(2); });
  document.getElementById('next-3').addEventListener('click', function(){
    var checked = document.querySelectorAll('input[name=reqs]checked');
    var notes = document.getElementById('reqs-other').value.trim();
